This runs Spark first and then Kafka producer/consumer processing. Kafka requires Docker services to be running:

$repo = (Get-Location).Path
docker run --rm `
	--name presight-spark `
	-v "${repo}:/opt/project" `
	-w /opt/project `
	apache/spark:4.0.1-python3 `
	/opt/spark/bin/spark-submit `
	solutions/submissions/adil_razack/03_big_data/spark_pipeline.py
